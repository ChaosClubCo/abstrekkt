  description: "Execute generated code in isolated VM. Returns stdout, stderr, exit code. Use before any generated code runs on real systems.",
  params: z.object({
    code: z.string(),
    language: z.enum(["typescript","python","javascript"])
  }),
  handler: async ({ code, language }) => {
    const sandbox = await csb.create({ template: language });
    const result  = await csb.exec(sandbox.id, code);
    await csb.destroy(sandbox.id);
    return { stdout: result.stdout, stderr: result.stderr, exit: result.exitCode };
  }
});
