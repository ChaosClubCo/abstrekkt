import type { ComponentType } from "react";
import * as CustomIcons from "./custom-icons";
import type { IconProps } from "./custom-icons";

export type IconName = 
  | "ink-quill"
  | "pulp-vat"
  | "botanical-press"
  | "alchemist-flask"
  | "sticker-sheet";

const ICON_REGISTRY: Record<IconName, ComponentType<IconProps>> = {
  "ink-quill": CustomIcons.InkQuill,
  "pulp-vat": CustomIcons.PulpVat,
  "botanical-press": CustomIcons.BotanicalPress,
  "alchemist-flask": CustomIcons.AlchemistFlask,
  "sticker-sheet": CustomIcons.StickerSheet,
} satisfies Record<IconName, ComponentType<IconProps>>;

export type AppIconProps = IconProps & {
  name: IconName;
};

export const AppIcon = ({ name, ...props }: AppIconProps) => {
  const IconComponent = ICON_REGISTRY[name];
  if (!IconComponent) return null;
  return <IconComponent {...props} />;
};
