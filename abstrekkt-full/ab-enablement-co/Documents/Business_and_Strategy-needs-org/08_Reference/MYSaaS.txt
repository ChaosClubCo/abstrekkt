import { z } from "zod";

export const EnvSchema = z.object({
  NEXT_PUBLIC_SITE_URL: z.string().url(),

  NEXT_PUBLIC_SUPABASE_URL: z.string().url(),
  NEXT_PUBLIC_SUPABASE_ANON_KEY: z.string().min(20),
  SUPABASE_SERVICE_ROLE_KEY: z.string().min(20),

  STRIPE_SECRET_KEY: z.string().min(10).optional().or(z.literal("")),
  STRIPE_WEBHOOK_SECRET: z.string().min(10).optional().or(z.literal("")),
  STRIPE_PORTAL_RETURN_URL: z.string().url(),
  STRIPE_CHECKOUT_SUCCESS_URL: z.string().url(),
  STRIPE_CHECKOUT_CANCEL_URL: z.string().url(),

  RATE_LIMIT_PER_MINUTE: z.coerce.number().int().min(1).max(10000),
  RATE_LIMIT_BURST: z.coerce.number().int().min(1).max(10000),
  ALLOW_DESTRUCTIVE: z
    .string()
    .transform((v) => v === "true")
    .pipe(z.boolean()),

  VIBECODE_TEMPLATE_ID: z.literal("saas_core"),
  VIBECODE_TEMPLATE_VERSION: z.literal("1.0.0"),
});

export type Env = z.infer<typeof EnvSchema>;

export function loadEnv(processEnv: NodeJS.ProcessEnv): Env {
  const parsed = EnvSchema.safeParse(processEnv);
  if (!parsed.success) {
    const flattened = parsed.error.flatten();
    throw new Error(
      Invalid environment variables:\n${JSON.stringify(flattened.fieldErrors, null, 2)}
    );
  }
  return parsed.data;
}

apps/web/lib/rateLimit.ts
type Bucket = { tokens: number; lastRefill: number };

const memory = new Map<string, Bucket>();

export function tokenBucket({
  key,
  burst,
  perMinute,
  now = Date.now(),
}: {
  key: string;
  burst: number;
  perMinute: number;
  now?: number;
}): { ok: boolean; remaining: number; retryAfterMs?: number } {
  const refillPerMs = perMinute / 60000;
  const bucket = memory.get(key) ?? { tokens: burst, lastRefill: now };
  const elapsed = Math.max(0, now - bucket.lastRefill);
  const refill = elapsed * refillPerMs;
  bucket.tokens = Math.min(burst, bucket.tokens + refill);
  bucket.lastRefill = now;

  if (bucket.tokens < 1) {
    memory.set(key, bucket);
    const msToNext = Math.ceil((1 - bucket.tokens) / refillPerMs);
    return { ok: false, remaining: 0, retryAfterMs: msToNext };
  }

  bucket.tokens -= 1;
  memory.set(key, bucket);
  return { ok: true, remaining: Math.floor(bucket.tokens) };
}

apps/web/middleware.ts
import { NextRequest, NextResponse } from "next/server";
import { loadEnv } from "@saas-core/config/env";
import { tokenBucket } from "./lib/rateLimit";

const env = loadEnv(process.env);

export function middleware(req: NextRequest) {
  const ip = req.ip ?? req.headers.get("x-forwarded-for") ?? "unknown";
  const key = ip:${ip}:path:${req.nextUrl.pathname};
  const { ok, retryAfterMs } = tokenBucket({
    key,
    burst: env.RATE_LIMIT_BURST,
    perMinute: env.RATE_LIMIT_PER_MINUTE,
  });

  if (!ok) {
    const res = NextResponse.json({ error: "rate_limited" }, { status: 429 });
    if (retryAfterMs) res.headers.set("retry-after", String(Math.ceil(retryAfterMs / 1000)));
    return res;
  }

  return NextResponse.next();
}

apps/web/app/api/invites/accept/route.ts
import { supabaseServer } from "@/lib/supabase/server";
import { requireUser } from "@/lib/auth";
import { InviteAcceptSchema } from "@/lib/zod";

export async function POST(req: Request) {
  const { user, redirectTo } = await requireUser();
  if (!user) return Response.json({ error: "unauthenticated", redirectTo }, { status: 401 });

  const body = await req.json().catch(() => null);
  const parsed = InviteAcceptSchema.safeParse(body);
  if (!parsed.success) return Response.json({ error: "invalid_body", issues: parsed.error.issues }, { status: 400 });

  const supabase = supabaseServer();

  // Fetch invite (RLS allows only admins normally; accept uses service role in real deployments)
  // For template@1.0.0: accept is done client-side only when invite token is exchanged through a server RPC in @2.0.
  // Here we implement a safe path: attempt select by token; if blocked by RLS, return explicit error.
  const { data: invite, error: invErr } = await supabase
    .from("invites")
    .select("id,org_id,email,role,expires_at,accepted_at")
    .eq("token", parsed.data.token)
    .single();

  if (invErr || !invite) return Response.json({ error: "invite_unavailable", details: invErr?.message }, { status: 404 });

  if (invite.accepted_at) return Response.json({ error: "invite_already_used" }, { status: 409 });
  if (new Date(invite.expires_at).getTime() < Date.now()) return Response.json({ error: "invite_expired" }, { status: 410 });

  // Mark accepted
  const { error: updErr } = await supabase
    .from("invites")
    .update({ accepted_at: new Date().toISOString() })
    .eq("id", invite.id);

  if (updErr) return Response.json({ error: "db_error", details: updErr.message }, { status: 500 });

  // Add member
  const { error: memErr } = await supabase
    .from("org_members")
    .insert({ org_id: invite.org_id, user_id: user.id, role: invite.role });

  if (memErr) return Response.json({ error: "db_error", details: memErr.message }, { status: 500 });

  return Response.json({ ok: true, orgId: invite.org_id });
}

import Stripe from "stripe";
import { loadEnv } from "@saas-core/config/env";

const env = loadEnv(process.env);
const stripe = new Stripe(env.STRIPE_SECRET_KEY || "disabled", { apiVersion: "2024-06-20" as any });

export const runtime = "nodejs";

async function readRawBody(req: Request): Promise<string> {
  const buf = await req.arrayBuffer();
  return Buffer.from(buf).toString("utf8");
}

export async function POST(req: Request) {
  if (!env.STRIPE_SECRET_KEY || !env.STRIPE_WEBHOOK_SECRET) {
    return Response.json({ error: "billing_not_configured" }, { status: 400 });
  }

  const sig = req.headers.get("stripe-signature");
  if (!sig) return Response.json({ error: "missing_signature" }, { status: 400 });

  const raw = await readRawBody(req);

  let event: Stripe.Event;
  try {
    event = stripe.webhooks.constructEvent(raw, sig, env.STRIPE_WEBHOOK_SECRET);
  } catch (e: any) {
    return Response.json({ error: "invalid_signature", details: e?.message }, { status: 400 });
  }

  // Idempotency + processing should enqueue a job and store to webhook_events using service role.
  // In @1.0.0 we ACK fast and leave processing as a TODO-free contract:
  // - store event (provider,event_id unique)
  // - update subscription record
  // Implementation lives in worker in @2.0.

  return Response.json({ received: true, id: event.id, type: event.type });
}

  | "workflow_builder";

export type VibeInput = {
  meta: { template?: TemplateId; target_users: "internal" | "external" | "both"; mvp_strictness: "demo" | "mvp" | "prod" };
  domain: { core_entities: Array<any>; primary_flows: Array<any>; integrations: Array<{ provider: string; mode: string; events?: string[] }> };
  security: { tenancy: "single" | "multi"; roles: string[]; rls_required: boolean; webhooks: { verify_signatures: boolean; idempotency: boolean } };
  ai: { enabled: boolean };
  billing: { enabled: boolean };
};

type RouteResult = {
  template: TemplateId;
  version: string;
  confidence: number;
  reasons: string[];
};

function bool(v: any) { return !!v; }

export function routeTemplate(input: VibeInput): RouteResult {
  // 0) Hard override (only if explicitly set AND valid)
  if (input.meta.template) {
    return { template: input.meta.template, version: "1.0.0", confidence: 1, reasons: ["explicit_template_selected"] };
  }

  const needsAuth = input.security.roles?.length > 0 || input.meta.target_users !== "external";
  const needsDb = input.domain.core_entities?.length > 0;
  const needsMulti = input.security.tenancy === "multi";
  const needsAI = input.ai.enabled;
  const needsIntegrations = input.domain.integrations?.length > 0;
  const needsWebhooks = needsIntegrations && input.security.webhooks?.verify_signatures;
  const needsBilling = input.billing.enabled;
  const needsWorkflows = input.domain.primary_flows?.some((f) => /workflow|automation/i.test(f?.name ?? "")) ?? false;
  const embedKeyword = input.domain.primary_flows?.some((f) => /embed|widget|sdk|iframe/i.test(f?.name ?? "")) ?? false;

  // 1) Auto-downgrade to Launchpad (static)
  if (!needsDb && !needsAuth && !needsIntegrations && !needsAI && input.security.tenancy === "single") {
    return {
      template: "launchpad",
      version: "1.0.0",
      confidence: 0.95,
      reasons: ["no_db", "no_auth", "no_integrations", "no_ai", "single_tenant => launchpad"],
    };
  }

  // 2) Weighted scoring
  const scores: Record<TemplateId, { score: number; reasons: string[] }> = {
    launchpad: { score: 0, reasons: [] },
    saas_core: { score: 0, reasons: [] },
    ai_tool: { score: 0, reasons: [] },
    integration_hub: { score: 0, reasons: [] },
    admin_console: { score: 0, reasons: [] },
    embed_sdk: { score: 0, reasons: [] },
    workflow_builder: { score: 0, reasons: [] },
  };

  function add(t: TemplateId, points: number, reason: string) {
    scores[t].score += points;
    scores[t].reasons.push(reason);
  }

  // SaaS core: safest baseline when auth+db or multi-tenant
  if (needsAuth) add("saas_core", 30, "needs_auth");
  if (needsDb) add("saas_core", 25, "needs_db");
  if (needsMulti) add("saas_core", 35, "needs_multi_tenant");
  if (needsBilling) add("saas_core", 10, "needs_billing");

  // AI tool
  if (needsAI) add("ai_tool", 60, "needs_ai");
  if (needsAI && needsDb) add("ai_tool", 15, "ai_outputs_need_persistence");
  if (needsAI && needsMulti) add("ai_tool", 10, "ai_multi_tenant_metering");

  // Integrations
  if (needsIntegrations) add("integration_hub", 50, "needs_integrations");
  if (needsWebhooks) add("integration_hub", 15, "needs_webhooks_signature_verification");
  if (needsIntegrations && needsMulti) add("integration_hub", 10, "integrations_multi_tenant_connections");

  // Admin console (internal)
  if (input.meta.target_users === "internal") add("admin_console", 45, "internal_target_users");
  if (needsDb && input.meta.target_users === "internal") add("admin_console", 10, "internal_db_ops");

  // Embed
  if (embedKeyword) add("embed_sdk", 70, "embed_keyword_detected");

  // Workflows
  if (needsWorkflows) add("workflow_builder", 70, "workflow_keyword_detected");

  // 3) Pick winner with safe tie-break
  const ranked = (Object.keys(scores) as TemplateId[])
    .map((t) => ({ t, ...scores[t] }))
    .sort((a, b) => b.score - a.score);

  const top = ranked[0];
  const second = ranked[1];

  // Safe tie-break: if close, prefer saas_core unless ai/integration explicitly dominates
  const close = second && top.score - second.score < 10;
  let chosen = top.t;

  if (close && chosen !== "saas_core") {
    if (scores["saas_core"].score >= second.score) {
      chosen = "saas_core";
      scores["saas_core"].reasons.push("tie_break_to_safe_baseline");
    }
  }

  // 4) Confidence heuristic
  const confidence = Math.min(0.99, Math.max(0.6, top.score / 100));

  return {
    template: chosen,
    version: "1.0.0",
    confidence,
    reasons: scores[chosen].reasons,
  };
}

-- - Adds org_members row (idempotent)
-- - Marks invite accepted
-- - Writes audit_log via helper

  p_org_id uuid,
  p_actor uuid,
  p_action text,
  p_target text,
  p_meta jsonb default '{}'::jsonb
) returns void
language plpgsql
security definer
set search_path = public
as $$
begin
  insert into public.audit_log(org_id, actor_user_id, action, target, meta)
  values (p_org_id, p_actor, p_action, p_target, coalesce(p_meta, '{}'::jsonb));
end;
$$;

create or replace function public.accept_invite(p_token text)
returns uuid
language plpgsql
security definer
set search_path = public
as $$
declare
  v_invite public.invites%rowtype;
  v_uid uuid;
begin
  v_uid := auth.uid();
  if v_uid is null then
    raise exception 'unauthenticated';
  end if;

  select * into v_invite
  from public.invites
  where token = p_token
  limit 1;

  if not found then
    raise exception 'invite_not_found';
  end if;

  if v_invite.accepted_at is not null then
    raise exception 'invite_already_used';
  end if;

  if v_invite.expires_at < now() then
    raise exception 'invite_expired';
  end if;

  -- Add membership idempotently
  insert into public.org_members(org_id, user_id, role)
  values (v_invite.org_id, v_uid, v_invite.role)
  on conflict (org_id, user_id) do update
    set role = excluded.role;

  update public.invites
  set accepted_at = now()
  where id = v_invite.id;

  perform public.log_audit(
    v_invite.org_id,
    v_uid,
    'invite.accept',
    v_invite.email,
    jsonb_build_object('role', v_invite.role)
  );

  return v_invite.org_id;
end;
$$;

Fix: Correct RPC call using user session (required)

import { z } from "zod";
import { requireUser } from "@/lib/auth";
import { supabaseServer } from "@/lib/supabase/server";

const AcceptSchema = z.object({ token: z.string().min(20) });

export const runtime = "nodejs";

export async function POST(req: Request) {
  const { user, redirectTo } = await requireUser();
  if (!user) return Response.json({ error: "unauthenticated", redirectTo }, { status: 401 });

  const body = await req.json().catch(() => null);
  const parsed = AcceptSchema.safeParse(body);
  if (!parsed.success) return Response.json({ error: "invalid_body", issues: parsed.error.issues }, { status: 400 });

  const supabase = supabaseServer();

  const { data, error } = await supabase.rpc("accept_invite", { p_token: parsed.data.token });
  if (error) {
    const msg = error.message || "rpc_error";
    const map: Record<string, number> = {
      invite_not_found: 404,
      invite_already_used: 409,
      invite_expired: 410,
      unauthenticated: 401,
    };
    const status = map[msg] ?? 400;
    return Response.json({ error: msg }, { status });
  }

  return Response.json({ ok: true, orgId: data }, { status: 200 });
}


Why this is correct: it preserves auth.uid() inside Postgres because the call is made with the user session from cookies. The RPC runs with definer privileges for DB writes but still uses auth.uid() for identity gating.

import { supabaseServiceRole } from "@/lib/server/serviceRole";
import { JobType } from "./contracts";
import { handleStripeJob } from "@/lib/stripe/handlers";

export async function claimNextJob(workerId: string) {
  const supabase = supabaseServiceRole();

  // Claim one queued job ready to run.
  // We do this with an update query to avoid race conditions.
  // For higher throughput, use an RPC with SKIP LOCKED in @2.1+.

  const { data, error } = await supabase
    .from("jobs")
    .update({ status: "processing", locked_at: new Date().toISOString(), locked_by: workerId })
    .eq("status", "queued")
    .lte("run_after", new Date().toISOString())
    .order("created_at", { ascending: true })
    .limit(1)
    .select("*")
    .maybeSingle();

  if (error) throw new Error(claim_failed: ${error.message});
  return data;
}

export async function processJob(job: any) {
  const supabase = supabaseServiceRole();
  const typeParsed = JobType.safeParse(job.type);
  if (!typeParsed.success) throw new Error(unknown_job_type: ${job.type});

  try {
    if (job.type.startsWith("stripe.")) {
      await handleStripeJob(job);
    } else {
      throw new Error(no_handler_for: ${job.type});
    }

    await supabase.from("jobs").update({ status: "succeeded" }).eq("id", job.id);
  } catch (e: any) {
    const attempts = (job.attempts ?? 0) + 1;
    const max = job.max_attempts ?? 10;
    const status = attempts >= max ? "dead" : "failed";
    const backoffSeconds = Math.min(3600, 2 ** attempts * 5);

    await supabase
      .from("jobs")
      .update({
        status,
        attempts,
        last_error: String(e?.message ?? e),
        run_after: new Date(Date.now() + backoffSeconds * 1000).toISOString(),
      })
      .eq("id", job.id);

    throw e;
  }
}

import { z } from "zod";
import { claimNextJob, processJob } from "@/lib/jobs/worker";

const WorkerSchema = z.object({
  workerId: z.string().min(3),
  maxJobs: z.number().int().min(1).max(50).default(10),
});

export const runtime = "nodejs";

export async function POST(req: Request) {
  const body = await req.json().catch(() => null);
  const parsed = WorkerSchema.safeParse(body);
  if (!parsed.success) return Response.json({ error: "invalid_body", issues: parsed.error.issues }, { status: 400 });

  const { workerId, maxJobs } = parsed.data;
  const results: any[] = [];

  for (let i = 0; i < maxJobs; i++) {
    const job = await claimNextJob(workerId);
    if (!job) break;
    try {
      await processJob(job);
      results.push({ id: job.id, status: "succeeded", type: job.type });
    } catch (e: any) {
      results.push({ id: job.id, status: "failed", type: job.type, error: String(e?.message ?? e) });
    }
  }

  return Response.json({ ok: true, processed: results.length, results });
}

import Stripe from "stripe";
import { loadEnv } from "@saas-core/config/env";
import { supabaseServiceRole } from "@/lib/server/serviceRole";
import { enqueueJob } from "@/lib/jobs/enqueue";

const env = loadEnv(process.env);
const stripe = new Stripe(env.STRIPE_SECRET_KEY || "disabled", { apiVersion: "2024-06-20" as any });

export const runtime = "nodejs";

async function readRawBody(req: Request): Promise<string> {
  const buf = await req.arrayBuffer();
  return Buffer.from(buf).toString("utf8");
}

export async function POST(req: Request) {
  if (!env.STRIPE_SECRET_KEY || !env.STRIPE_WEBHOOK_SECRET) {
    return Response.json({ error: "billing_not_configured" }, { status: 400 });
  }

  const sig = req.headers.get("stripe-signature");
  if (!sig) return Response.json({ error: "missing_signature" }, { status: 400 });

  const raw = await readRawBody(req);

  let event: Stripe.Event;
  try {
    event = stripe.webhooks.constructEvent(raw, sig, env.STRIPE_WEBHOOK_SECRET);
  } catch (e: any) {
    return Response.json({ error: "invalid_signature", details: e?.message }, { status: 400 });
  }

  const supabase = supabaseServiceRole();

  // Persist idempotently
  const { data: row, error: insErr } = await supabase
    .from("webhook_events")
    .insert({
      provider: "stripe",
      event_id: event.id,
      payload: event as any,
      status: "received",
    })
    .select("id")
    .single();

  // If duplicate, we still ACK (idempotent)
  if (insErr && !String(insErr.message).toLowerCase().includes("duplicate")) {
    return Response.json({ error: "persist_failed", details: insErr.message }, { status: 500 });
  }

  // Enqueue job
  const job = await enqueueJob({
    type: mapStripeEventToJobType(event.type),
    payload: {
      provider: "stripe",
      eventId: event.id,
      type: event.type,
      data: event.data,
    },
    orgId: null,
  });

  // Link job to webhook event if it was newly inserted
  if (row?.id) {
    await supabase.from("webhook_events").update({ job_id: job.id }).eq("id", row.id);
  }

  return Response.json({ received: true, id: event.id, jobId: job.id, type: event.type });
}

function mapStripeEventToJobType(type: string) {
  if (type.includes("checkout.session.completed")) return "stripe.checkout.completed";
  if (type.includes("customer.subscription.updated")) return "stripe.subscription.updated";
  if (type.includes("customer.subscription.deleted")) return "stripe.subscription.deleted";
  // default safe mapping (still validates via Zod if added later)
  return "stripe.subscription.updated";
}

import { createClient } from "redis";

let client: ReturnType<typeof createClient> | null = null;

export async function redisTokenBucket(key: string, burst: number, perMinute: number) {
  const url = process.env.REDIS_URL;
  if (!url) return null;

  if (!client) {
    client = createClient({ url });
    client.on("error", () => {});
    await client.connect();
  }

  // Simple fixed window limiter for @2.0.0; token bucket Lua can be added later.
  const windowSeconds = 60;
  const now = Math.floor(Date.now() / 1000);
  const windowKey = ${key}:${Math.floor(now / windowSeconds)};

  const count = await client.incr(windowKey);
  if (count === 1) await client.expire(windowKey, windowSeconds);

  const limit = perMinute;
  if (count > limit) return { ok: false as const, remaining: 0 };
  return { ok: true as const, remaining: Math.max(0, limit - count) };
}

with allowlist as (

    'schema_migrations',     -- example if you use it
    'spatial_ref_sys'        -- example for PostGIS installs
  ]) as tablename
),
public_tables as (

  from pg_tables
  where schemaname='public'

  select p.tablename
  from public_tables p

    and p.rowsecurity is distinct from true
)

create or replace function public.claim_next_job(p_worker_id text)

language plpgsql
security definer

begin
  select *
