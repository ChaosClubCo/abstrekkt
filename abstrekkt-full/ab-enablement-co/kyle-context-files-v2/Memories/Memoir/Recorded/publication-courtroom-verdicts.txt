const App = () => {
  const [expanded, setExpanded] = React.useState(null);
  const [filterTier, setFilterTier] = React.useState('all');
  const filtered = filterTier === 'all' ? publications : publications.filter(p => p.tier === filterTier);
  const verdicts = {
    viable: publications.filter(p => p.verdict.includes('VIABLE')).length,
    conditional: publications.filter(p => p.verdict.includes('CONDITIONAL')).length,
    watch: publications.filter(p => p.verdict.includes('WATCH')).length
  };
  const tier1 = publications.filter(p => p.tier === 'tier1');
  return (
        {/* HEADER */}
          Publication Courtroom
          Kyle Rosebrook | Memoir Essays | 5-Role Debates | Saturation Strategy
          {/* VERDICT SUMMARY */}
              Ready to Shoot
              {verdicts.viable}
              Viable targets
              Conditional
              {verdicts.conditional}
              Verify fit first
              Strategy
              Week 1-3
              Full saturation plan
          {/* WEEK 1-2 PRIORITY */}
            ⚡ WEEK 1-2: TIER-1 SIMULTANEOUS BLITZ
              {tier1.map((pub, idx) => (
                  #{idx + 1} {pub.name}
                  {pub.verdict.split(' - ')[1]}
              ))}
            Strategy: Submit all simultaneously. Different routes (pitches vs. direct) = no conflict. Expect decisions 14-30 days.
          {/* FILTERS */}
             setFilterTier('all')} className={`px-4 py-2 rounded-lg font-semibold transition ${filterTier === 'all' ? 'bg-blue-600 text-white' : 'bg-gray-200 text-gray-700'}`}>All (10 shown)
             setFilterTier('tier1')} className={`px-4 py-2 rounded-lg font-semibold transition ${filterTier === 'tier1' ? 'bg-green-600 text-white' : 'bg-gray-200 text-gray-700'}`}>Tier 1 (7)
             setFilterTier('tier2')} className={`px-4 py-2 rounded-lg font-semibold transition ${filterTier === 'tier2' ? 'bg-green-600 text-white' : 'bg-gray-200 text-gray-700'}`}>Tier 2 (3)
        {/* PUBLICATIONS */}
          {filtered.map(pub => {
            const isGreen = pub.verdict.includes('VIABLE') && !pub.verdict.includes('WATCH');
            const isYellow = pub.verdict.includes('CONDITIONAL');
            const borderColor = isGreen ? 'border-green-400' : isYellow ? 'border-yellow-400' : 'border-blue-400';
            const bgColor = isGreen ? 'bg-green-50' : isYellow ? 'bg-yellow-50' : 'bg-blue-50';
            return (
                 setExpanded(expanded === pub.id ? null : pub.id)} className="w-full p-5 flex justify-between items-start hover:bg-opacity-75 transition text-left">
                      {pub.name}
                      {pub.category}
                    {pub.contact}
                    {pub.rationale}
                    {pub.verdict}
                {/* EXPANDED */}
                {expanded === pub.id && (
                    {/* ROLES */}
                      5-Role Debate
                        {Object.entries(pub.roles).map(([roleName, role]) => (
                              {roleName === 'publicist' ? '🎤' : roleName === 'legal' ? '⚖️' : roleName === 'marketing' ? '📊' : roleName === 'assistant' ? '📋' : '🤝'} {roleName}
                                {role.stance}
                            {role.text}
                        ))}
                    {/* METADATA */}
                        Payment
                        {pub.payment}
                        Response
                        {pub.response}
                        Timeline
                        {pub.timeline}
                    {/* NEXT STEPS */}
                      → NEXT STEPS
                        {pub.nextSteps.map((step, idx) => (
                            •
                            {step}
                        ))}
                )}
            );
          })}
        {/* TOP 3 UNKNOWNS */}
        {filterTier === 'all' && (
              🚨 TOP 3 UNKNOWNS (BEFORE FIRST PITCH)
                1. LEGAL REVIEW: Intinc Boundaries
                The risk: Non-competes + IP agreements vary. Publishing might violate employment contract.
                What to do: Get tech employment lawyer (~$200-300). Ask: (1) Can I publish about this while employed? (2) What details are off-limits? (3) Want legal review of final draft?
                Skip it: Cease-and-desist, forced unpublication. Do it: Peace of mind.
                2. RIGHTS STRATEGY: What's Your Endgame?
                The question: Republish everywhere? Exclusive first-pub to tier-1 then open tier-2/3? Customize per outlet?
                Why: Atlantic vs. Longreads vs. Substack all want different rights. Different answer = different pitch approach.
                Recommendation: Republish everywhere (saturation = higher placement chance). Clarify BEFORE first submission.
                3. PITCH VARIATIONS: Different Angles Per Outlet
                Problem: Atlantic wants "construction vs. recovery" philosophically. MIT TR wants "systems thinking" technically. WIRED wants "unconventional engineer" culturally. Same essay, different pitches.
                Create: 5 pitch variations (literary, tech-culture, tech-systems, business, AI-future).
                Generic pitches: Low open rates. Specific pitches: Each outlet sees why essay is perfect *for them*.
            {/* EXECUTION CHECKLIST */}
              ✅ Execution Checklist
                  Week 0 (NOW)
                    □ Legal review scheduled
                    □ Decide rights strategy
                    □ Draft 5 pitch variations
                  Week 1-2
                    □ Send all tier-1 pitches/submissions
                    □ Expect responses 14-30 days
                    □ Track in spreadsheet
                  Week 2-3
                    □ Submit tier-2 outlets
                    □ Prepare for revisions
                  Week 3+
                    □ Saturate tier-3 (15+ more)
                    □ Track all acceptances
        )}
  );
};
