<div id="practices" class="section">
  <h2 style="font-size: 18px; margin: 0 0 1.5rem 0;">Suno v5 Best Practices</h2>
  
  <div class="practice-item">
    <p class="practice-title">1. Custom Mode, Not Simple</p>
    <p class="practice-desc">Custom Mode gives full control. Simple mode is too generic.</p>
  </div>
  
  <div class="practice-item">
    <p class="practice-title">2. Avoid Artist Names</p>
    <p class="practice-desc">Use style descriptors: "sparse folk confessional" not "sound like Taylor Swift."</p>
  </div>
  
  <div class="practice-item">
    <p class="practice-title">3. Specific Instrumentation</p>
    <p class="practice-desc">Name exact instruments. Acoustic guitar fingerpicking, 808 bass, cello—not "synths."</p>
  </div>
  
  <div class="practice-item">
    <p class="practice-title">4. Set BPM and Key Explicitly</p>
    <p class="practice-desc">State tempo (65 BPM) and key (E minor, G major). Anchors the generation.</p>
  </div>
  
  <div class="practice-item">
    <p class="practice-title">5. Describe Vocal Character in Detail</p>
    <p class="practice-desc">Not just "male voice." Say "weathered male vocals, contemplative but not performative."</p>
  </div>
  
  <div class="practice-item">
    <p class="practice-title">6. Generate 2-3 Variations Per Song</p>
    <p class="practice-desc">Suno often needs multiple attempts. Generate at least 2 versions.</p>
  </div>
  
  <div class="practice-item">
    <p class="practice-title">7. Iterate Based on Output</p>
    <p class="practice-desc">If beat is too bouncy, add "tight, measured" to description. If vocals sound AI, specify "organic delivery."</p>
  </div>
  
  <div class="practice-item">
    <p class="practice-title">8. Export Stems Via Suno Studio</p>
    <p class="practice-desc">Suno Studio exports up to 12 WAV stems for DAW polish.</p>
  </div>
  
  <div class="practice-item">
    <p class="practice-title">9. Always Use v5</p>
    <p class="practice-desc">v5 is current and studio-quality. Regenerate from scratch with v5.</p>
  </div>

  <div class="practice-item">
    <p class="practice-title">10. Expected Timeline</p>
    <p class="practice-desc">30-60 seconds per track. 3 versions = ~3 minutes. Iterate 2-3x = ~10 minutes total.</p>
  </div>

  <div class="practice-item">
    <p class="practice-title">11. Use Your Own Vocal Samples (No Rerecording)</p>
    <p class="practice-desc">Upload 5-30 second audio sample of your voice. Suno uses it as a style reference. Your accent, cadence, and delivery influence output without rerecording.</p>
  </div>

  <div style="margin-top: 2rem; padding: 1.5rem; background: #f5f5f5; border: 1px solid #e0e0e0; border-radius: 12px;">
    <p style="font-weight: 500; margin: 0 0 1rem 0; font-size: 14px;">HOW TO USE YOUR VOCAL SAMPLES</p>
    
    <p style="font-size: 13px; margin: 0 0 1rem 0;"><strong>Step 1: Prepare Sample</strong><br>Format: WAV, MP3, OGG. Length: 5-30 seconds. Content: singing, spoken word, humming—just your voice character.</p>
    
    <p style="font-size: 13px; margin: 0 0 1rem 0;"><strong>Step 2: Suno Studio → Custom Mode</strong><br>Look for "Audio Upload" or "Reference Audio" field below lyrics input.</p>
    
    <p style="font-size: 13px; margin: 0 0 1rem 0;"><strong>Step 3: Upload Your Sample</strong><br>Click upload → Select audio file. Suno previews it.</p>
    
    <p style="font-size: 13px; margin: 0 0 1rem 0;"><strong>Step 4: Set Audio Influence Slider</strong><br>Low (10-30%): Subtle hint. Medium (40-60%): Sweet spot. High (70-100%): Heavy influence.</p>
    
    <p style="font-size: 13px; margin: 0 0 1rem 0;"><strong>Step 5: Fill Production + Paste Lyrics</strong><br>Provide production brief + song lyrics with [Verse]/[Chorus]/[Bridge] tags.</p>

function switchTab(tabName) {
  const sections = document.querySelectorAll('.section');
  sections.forEach(s => s.classList.remove('active'));
  document.getElementById(tabName).classList.add('active');
  
  const tabs = document.querySelectorAll('.nav-tab');
  tabs.forEach(t => t.classList.remove('active'));
  event.target.classList.add('active');
}

function toggleCandidate(card) {
  const details = card.querySelector('.candidate-details');
  details.classList.toggle('expanded');
  const toggle = card.querySelector('span:last-child');
  toggle.textContent = details.classList.contains('expanded') ? '−' : '+';
}

function switchSongTab(button, tabId) {
  const tabs = button.parentElement.querySelectorAll('.tab-button');
  tabs.forEach(t => t.classList.remove('active'));
  button.classList.add('active');
  
  const tabsContainer = button.parentElement.parentElement;
  const contents = tabsContainer.querySelectorAll('.tab-content');
  contents.forEach(c => c.classList.remove('active'));
  document.getElementById(tabId).classList.add('active');
}

function copyToClipboard(text, button) {
  navigator.clipboard.writeText(text).then(() => {
    button.classList.add('copied');
    button.textContent = 'Copied!';
    setTimeout(() => {
      button.classList.remove('copied');
      button.textContent = 'Copy ' + (text.includes('[Verse]') ? 'Lyrics' : (text.includes('BPM') ? 'Prompt' : 'Notes'));
    }, 2000);
  });
}
