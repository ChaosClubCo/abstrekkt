# =============================================================================
# RAPID SYSTEM CHECK
# =============================================================================

quick_checks() {
    log "🔍 Running quick system checks..."
    
    # Check OS
    if [[ "$OSTYPE" != "linux-gnu"* ]] && [[ "$OSTYPE" != "darwin"* ]]; then
        error "Unsupported OS. This installer supports Linux and macOS."
    fi
    
    # Check required commands
    local missing=()
    for cmd in curl bash mkdir chmod; do
        if ! command -v "$cmd" &> /dev/null; then
            missing+=("$cmd")
        fi
    done
    
    if [ ${#missing[@]} -gt 0 ]; then
        error "Missing required commands: ${missing[*]}"
    fi
    
    # Check disk space (need 2GB minimum)
    local available_gb
    if command -v df &> /dev/null; then
        available_gb=$(df ~ 2>/dev/null | awk 'NR==2 {print int($4/1024/1024)}' || echo "10")
        if [ "$available_gb" -lt 2 ]; then
            echo -e "${YELLOW}⚠️  Low disk space: ${available_gb}GB available${NC}"
        fi
    fi
    
    log "✅ System checks passed"
}

# =============================================================================
# INSTALL PREREQUISITES
# =============================================================================

install_prerequisites() {
    log "📦 Installing prerequisites..."
    
    # Detect package manager and install essentials
    if command -v apt-get &> /dev/null; then
        # Ubuntu/Debian
        sudo apt-get update -qq
        sudo apt-get install -y curl git jq openssl
    elif command -v yum &> /dev/null; then
        # RHEL/CentOS
        sudo yum install -y curl git jq openssl
    elif command -v brew &> /dev/null; then
        # macOS
        brew install jq openssl
    elif command -v apk &> /dev/null; then
        # Alpine
        apk add --no-cache curl git jq openssl
    else
        log "⚠️  No package manager detected. Assuming prerequisites are installed."
    fi
    
    log "✅ Prerequisites installed"
}

# =============================================================================
# SECURITY SYSTEM
# =============================================================================

setup_security() {
    log "🔒 Setting up enterprise security..."
    
    # Generate CA and certificates
    openssl genrsa -out "$AGENT_HOME/security/ca.key" 4096 2>/dev/null
    openssl req -new -x509 -key "$AGENT_HOME/security/ca.key" \
        -sha256 -subj "/C=US/ST=CA/O=AgentSystem/CN=AgentCA" \
        -days 3650 -out "$AGENT_HOME/security/ca.crt" 2>/dev/null
    
    # Generate system encryption keys
    openssl rand -hex 32 > "$AGENT_HOME/security/system.key"
    chmod 600 "$AGENT_HOME/security"/*
    
    # Create RBAC configuration
    cat > "$AGENT_HOME/permissions/rbac.yaml" << 'EOF'
roles:
  supervisor:
    name: "tessa"
    permissions: ["*"]
  specialist:
    permissions: ["code:read", "code:write", "deploy:staging"]
  security:
    permissions: ["security:audit", "compliance:check"]
access_matrix:
  production_deployment:
    required_approvals: 2
    required_roles: ["supervisor", "security"]
EOF
    
    log "✅ Security system configured"
}

# DOCKER DEPLOYMENT SETUP
# =============================================================================

setup_docker_deployment() {
    log "🐳 Setting up Docker deployment..."
    
    cat > "$AGENT_HOME/deployments/docker-compose.yml" << 'EOF'
version: '3.8'
services:
  tessa:
    image: alpine:latest
    container_name: agent-tessa
    command: sh -c "echo 'Tessa supervisor running...' && sleep infinity"
    environment:
      - AGENT_NAME=tessa
      - AGENT_ROLE=supervisor
    ports:
      - "8080:8080"
    volumes:
      - ./logs:/var/log/agents
    restart: unless-stopped
    
  redis:
    image: redis:7-alpine
    container_name: agent-redis
    command: redis-server --requirepass agentpass
    volumes:
      - redis_data:/data
    restart: unless-stopped
    
  monitoring:
    image: prom/prometheus:latest
    container_name: agent-prometheus
    ports:
      - "9090:9090"
    restart: unless-stopped

volumes:
  redis_data:
EOF
    
    # Create deployment script
    cat > "$AGENT_HOME/bin/deploy" << 'EOF'
#!/bin/bash
cd "$AGENT_HOME/deployments"
if command -v docker-compose &> /dev/null; then
    echo "🚀 Starting agent system with Docker..."
    docker-compose up -d
    echo "✅ System deployed! Access at: http://localhost:8080"
else
    echo "⚠️  Docker Compose not found. Install Docker to use deployment features."
fi
EOF
    chmod +x "$AGENT_HOME/bin/deploy"
    
    log "✅ Docker deployment ready"
}

# =============================================================================
# CLI AND MANAGEMENT TOOLS
# =============================================================================

create_management_cli() {
    log "⚙️  Creating management CLI..."
    
    cat > "$AGENT_HOME/bin/agents" << 'EOF'
#!/bin/bash
AGENT_HOME="${AGENT_HOME:-$HOME/.agents}"

case "${1:-help}" in
    "status")
        echo "🤖 Agent System Status"
        echo "======================"
        echo "Home: $AGENT_HOME"
        echo "Agents: $(find "$AGENT_HOME/config" -name "*.json" | wc -l)"
        echo "Active: $(find "$AGENT_HOME/agents" -name "*.pid" | wc -l)"
        echo ""
        echo "Available Agents:"
        find "$AGENT_HOME/config" -name "*.json" -exec basename {} .json \; | sort
        ;;
    "start")
        agent_name="${2:-all}"
        if [ "$agent_name" = "all" ]; then
            echo "🚀 Starting all agents..."
            find "$AGENT_HOME/config" -name "*.json" -exec basename {} .json \; | while read agent; do
                if [ -f "$AGENT_HOME/agents/$agent/start.sh" ]; then
                    bash "$AGENT_HOME/agents/$agent/start.sh" &
                    echo "Started: $agent"
                fi
            done
        else
            if [ -f "$AGENT_HOME/agents/$agent_name/start.sh" ]; then
                bash "$AGENT_HOME/agents/$agent_name/start.sh" &
                echo "Started: $agent_name"
            else
                echo "Agent not found: $agent_name"
            fi
        fi
        ;;
    "stop")
        echo "🛑 Stopping agents..."
        find "$AGENT_HOME/agents" -name "*.pid" | while read pidfile; do
            if [ -f "$pidfile" ]; then
                pid=$(cat "$pidfile")
                kill "$pid" 2>/dev/null && echo "Stopped: $(basename "$pidfile" .pid)"
                rm -f "$pidfile"
            fi
        done
        ;;
    "deploy")
        "$AGENT_HOME/bin/deploy"
        ;;
    "logs")
        agent_name="${2:-system}"
        log_file="$AGENT_HOME/logs/${agent_name}.log"
        if [ -f "$log_file" ]; then
            tail -f "$log_file"
        else
            echo "Log not found. Available logs:"
            ls -1 "$AGENT_HOME/logs/" 2>/dev/null || echo "No logs yet"
        fi
        ;;
    "delegate")
        supervisor="${2:-tessa}"
        task="${3:-status check}"
        echo "🎯 Delegating task to $supervisor: $task"
        echo "Task delegation logged at $(date)" >> "$AGENT_HOME/logs/delegation.log"
        ;;
    *)
        echo "🤖 Agent Management CLI"
        echo ""
        echo "Usage: agents <command> [options]"
        echo ""
        echo "Commands:"
        echo "  status              Show system status"
        echo "  start [agent|all]   Start agent(s)"  
        echo "  stop                Stop all agents"
        echo "  deploy              Deploy with Docker"
        echo "  logs [agent]        View agent logs"
        echo "  delegate <task>     Delegate task to supervisor"
        echo ""
        echo "Examples:"
        echo "  agents status"
        echo "  agents start tessa"
        echo "  agents delegate 'build e-commerce platform'"
        ;;
esac
EOF
    chmod +x "$AGENT_HOME/bin/agents"
    
    log "✅ Management CLI created"
}

# =============================================================================
# SHELL INTEGRATION
# =============================================================================

setup_shell_integration() {
    log "🔧 Setting up shell integration..."
    
    # Determine shell and add to PATH
    local shell_rc=""
    if [ -n "${BASH_VERSION:-}" ]; then
        shell_rc="$HOME/.bashrc"
    elif [ -n "${ZSH_VERSION:-}" ]; then
        shell_rc="$HOME/.zshrc"
    fi
    
    if [ -n "$shell_rc" ] && [ -f "$shell_rc" ]; then
        if ! grep -q "AGENT_HOME" "$shell_rc" 2>/dev/null; then
            {
                echo ""
                echo "# Agent System - Added by one-command installer"
                echo "export AGENT_HOME=\"$AGENT_HOME\""
                echo "export PATH=\"\$PATH:\$AGENT_HOME/bin\""
                echo "alias agent-status='agents status'"
                echo "alias agent-start='agents start'"
                echo "alias agent-deploy='agents deploy'"
            } >> "$shell_rc"
        fi
    fi
    
    # Also create a temporary PATH setup for immediate use
    export PATH="$PATH:$AGENT_HOME/bin"
    
    log "✅ Shell integration configured"
}

# =============================================================================
# FINAL SYSTEM VALIDATION
# =============================================================================

validate_installation() {
    log "✅ Validating installation..."
    
    local validation_errors=0
    
    # Check directories
    local required_dirs=("config" "permissions" "agents" "security" "bin")
    for dir in "${required_dirs[@]}"; do
        if [ ! -d "$AGENT_HOME/$dir" ]; then
            echo "❌ Missing directory: $dir"
            ((validation_errors++))
        fi
    done
    
    # Check agents
    local agent_count=$(find "$AGENT_HOME/config" -name "*.json" | wc -l)
    if [ "$agent_count" -lt 15 ]; then
        echo "❌ Expected 20+ agents, found: $agent_count"
        ((validation_errors++))
    fi
    
    # Check CLI
    if [ ! -x "$AGENT_HOME/bin/agents" ]; then
        echo "❌ CLI tool not executable"
        ((validation_errors++))
    fi
    
    if [ "$validation_errors" -eq 0 ]; then
        log "✅ Installation validation passed"
        return 0
    else
        log "❌ Validation failed with $validation_errors errors"
        return 1
    fi
}
